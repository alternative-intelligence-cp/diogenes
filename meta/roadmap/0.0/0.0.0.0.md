### Step 1: write the probe files (executor, Write tool, into `<P>`)

`<P>/take.py`:

```python
import os, sqlite3, sys, time
me, mode, seconds, db = sys.argv[1], sys.argv[2], float(sys.argv[3]), sys.argv[4]
d = os.path.dirname(os.path.abspath(db))
ns = ' '.join(os.readlink(f'/proc/self/ns/{n}') for n in ('net', 'mnt', 'pid'))
t0 = time.time()
while not os.path.exists(os.path.join(d, 'go.start')):          # one start signal for every racer
    if time.time() > t0 + 120: print(me, 'no go.start within 120 s'); sys.exit(1)
    time.sleep(0.02)
wins = rounds = errors = 0; first = last = None; end = time.time() + seconds
def take(ttl):
    global first, last
    c = sqlite3.connect(db, timeout=5, isolation_level=None)
    c.execute('PRAGMA busy_timeout=5000'); c.execute('BEGIN IMMEDIATE')
    t = time.time()
    r = c.execute("UPDATE lease SET holder=?, fence=fence+1, expires_at=? WHERE id='t' AND (holder IS NULL OR expires_at<?)",
                  (me, t + ttl, t)).rowcount
    c.execute('COMMIT'); c.close()
    first = t if first is None else first; last = t
    return r
if mode == 'free':
    while time.time() < end:
        try: wins += take(0.001)
        except sqlite3.OperationalError as e: errors += 1; print('ERR', e, flush=True)
        rounds += 1
else:
    while time.time() < end:
        rounds += 1
        while not os.path.exists(os.path.join(d, f'go.{rounds}')):
            if time.time() > end: break
            time.sleep(0.005)
        if time.time() > end: rounds -= 1; break
        try: wins += take(60)
        except sqlite3.OperationalError as e: errors += 1; print('ERR', e, flush=True)
fence = sqlite3.connect(db, timeout=5).execute("SELECT fence FROM lease WHERE id='t'").fetchone()[0]
print(me, mode, 'rounds', rounds, 'wins', wins, 'errors', errors, 'fence', fence,
      'first', '-' if first is None else f'{first:.3f}', 'last', '-' if last is None else f'{last:.3f}', 'ns', ns)
```

`<P>/rounds.sh` (the referee, host side):

DB=/home/randy/.local/share/diogenes-probe-db; cd "$DB"; i=1; end=$(( $(date +%s) + $1 )); touch go.start
